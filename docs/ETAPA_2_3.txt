Idempotencia:
Una sección crítica RLock por servicio engloba búsqueda, creación y publicación de
pedido y clave. En la aplicación existe un singleton compartido por las rutas.
Misma clave y mismos IDs/cantidades (independiente del orden de items): pedido original.
Misma clave y distinto contenido válido: 409. Claves distintas: UUIDs de pedido distintos.
Un reintento de un pedido vencido recupera ese pedido como expired; no crea otro.
El almacenamiento es memoria local: no resiste reinicios, no coordina múltiples
procesos o réplicas y no tiene limpieza automática. Producción requiere persistencia
transaccional con unicidad para la clave y una política de retención.

"""
Punto de entrada de la aplicación Flask para Club VOID.
"""

import os
from flask import Flask, jsonify
from flask_cors import CORS
from werkzeug.exceptions import HTTPException

from config import get_config
from routes.health import health_bp
from routes.orders import orders_bp
from routes.products import products_bp


def create_app(config_name: str = None) -> Flask:
    app = Flask(__name__)

    config_class = get_config(config_name)
    app.config.from_object(config_class)

    app.json.ensure_ascii = False
    app.json.sort_keys = False

    # Limitar tamaño máximo de solicitudes (defensa básica).
    app.config["MAX_CONTENT_LENGTH"] = 64 * 1024

    CORS(
        app,
        resources={r"/api/*": {"origins": app.config["CORS_ORIGINS"]}},
        supports_credentials=False,
    )

    app.register_blueprint(health_bp, url_prefix=app.config["API_PREFIX"])
    app.register_blueprint(products_bp, url_prefix=app.config["API_PREFIX"])
    app.register_blueprint(orders_bp, url_prefix=app.config["API_PREFIX"])

    register_error_handlers(app)

    @app.route("/", methods=["GET"])
    def index():
        return jsonify(
            {
                "name": "Club VOID API",
                "version": app.config["API_VERSION"],
                "status": "running",
            }
        ), 200

    return app


def register_error_handlers(app: Flask) -> None:

    @app.errorhandler(400)
    def bad_request(error):
        return jsonify(
            {"error": "bad_request", "message": "La solicitud es inválida.", "status_code": 400}
        ), 400

    @app.errorhandler(404)
    def not_found(error):
        return jsonify(
            {"error": "not_found", "message": "El recurso solicitado no existe.", "status_code": 404}
        ), 404

    @app.errorhandler(405)
    def method_not_allowed(error):
        return jsonify(
            {"error": "method_not_allowed", "message": "El método HTTP no está permitido para este recurso.", "status_code": 405}
        ), 405

    @app.errorhandler(413)
    def payload_too_large(error):
        return jsonify(
            {"error": "payload_too_large", "message": "El cuerpo de la solicitud excede el tamaño máximo.", "status_code": 413}
        ), 413

    @app.errorhandler(500)
    def internal_error(error):
        return jsonify(
            {"error": "internal_server_error", "message": "Ocurrió un error interno en el servidor.", "status_code": 500}
        ), 500

    @app.errorhandler(Exception)
    def unhandled_exception(error):
        if isinstance(error, HTTPException):
            return jsonify(
                {"error": error.name.lower().replace(" ", "_"),
                 "message": error.description, "status_code": error.code}
            ), error.code
        app.logger.error("Error interno no manejado en la API.")
        return jsonify(
            {"error": "internal_server_error", "message": "Ocurrió un error interno en el servidor.", "status_code": 500}
        ), 500


app = create_app()


if __name__ == "__main__":
    host = os.getenv("FLASK_HOST", "127.0.0.1")
    port = int(os.getenv("FLASK_PORT", "5000"))
    app.run(host=host, port=port, debug=app.config["DEBUG"])

"""
Blueprint de pedidos de demostración.
"""

from flask import Blueprint, jsonify, request
from werkzeug.exceptions import RequestEntityTooLarge

from services.order_service import (
    IdempotencyConflictError,
    order_service,
)
from utils.validators import ValidationError, validate_idempotency_key

orders_bp = Blueprint("orders", __name__)

MAX_BODY_BYTES = 8 * 1024  # 8 KB


@orders_bp.route("/orders", methods=["POST"])
def create_order():
    raw_length = request.content_length or 0
    if raw_length > MAX_BODY_BYTES:
        return (
            jsonify(
                {
                    "error": "payload_too_large",
                    "message": "El cuerpo de la solicitud excede el tamaño máximo.",
                    "status_code": 413,
                }
            ),
            413,
        )

    try:
        idempotency_key = validate_idempotency_key(
            request.headers.get("Idempotency-Key")
        )
    except ValidationError as exc:
        return _validation_response(exc)

    # Comprobar también los bytes reales: Content-Length puede faltar.
    # Flask limita esta lectura a 64 KB a nivel global.
    if len(request.get_data(cache=True)) > MAX_BODY_BYTES:
        raise RequestEntityTooLarge()
    payload = request.get_json(silent=True)

    try:
        order, created = order_service.create_order(payload, idempotency_key)
    except ValidationError as exc:
        return _validation_response(exc)
    except IdempotencyConflictError as exc:
        return (
            jsonify(
                {
                    "error": "idempotency_conflict",
                    "message": str(exc),
                    "status_code": 409,
                }
            ),
            409,
        )

    status_code = 201 if created else 200
    return jsonify(order.to_dict()), status_code


@orders_bp.route("/orders/<order_id>", methods=["GET"])
def get_order(order_id: str):
    order = order_service.get_order(order_id)
    if order is None:
        return (
            jsonify(
                {
                    "error": "not_found",
                    "message": "El pedido solicitado no existe.",
                    "status_code": 404,
                }
            ),
            404,
        )
    return jsonify(order.to_dict()), 200


def _validation_response(exc: ValidationError):
    payload = exc.to_dict()
    payload["status_code"] = 400
    return jsonify(payload), 400

"""
Servicio de pedidos de demostración.

Responsable de:
- Validar contenido.
- Calcular importes desde el catálogo del backend.
- Crear pedidos temporales con idempotencia.
- Almacenar en memoria.
"""

import uuid
from copy import deepcopy
from threading import RLock
from datetime import datetime, timedelta, timezone
from typing import Dict, Optional, Tuple

from data.product_catalog import get_product
from models.order import Order, OrderItem
from services.idempotency_store import IdempotencyStore, idempotency_store
from utils.money import compute_subtotal
from utils.validators import ValidationError, validate_payload

SERVICE_FEE_CLP = 0
QUOTE_TTL_MINUTES = 15


class IdempotencyConflictError(Exception):
    """La misma clave se usó con contenido distinto."""


class OrderNotFoundError(Exception):
    """Pedido inexistente."""


class OrderService:
    def __init__(self, store=None) -> None:
        self._orders: Dict[str, Order] = {}
        self._store = store if store is not None else IdempotencyStore()
        self._lock = RLock()

    def create_order(self, payload, idempotency_key: str) -> Tuple[Order, bool]:
        """
        Crea o recupera un pedido.

        Devuelve (order, created) donde created indica si fue nuevo.
        Lanza IdempotencyConflictError si la clave ya existe con distinto
        contenido.
        """
        items = validate_payload(payload)
        fingerprint = self._build_fingerprint(items)

        # Una sola sección crítica incluye lectura, creación y publicación.
        # Ningún hilo puede observar una clave cuyo pedido aún no existe.
        with self._lock:
            existing = self._store.get(idempotency_key)
            if existing is not None:
                existing_fingerprint, order_id = existing
                if existing_fingerprint != fingerprint:
                    raise IdempotencyConflictError(
                        "La Idempotency-Key ya fue usada con un contenido distinto."
                    )
                return self.get_order(order_id), False

            order = self._build_order(items, idempotency_key)
            self._orders[order.order_id] = order
            self._store.try_register(idempotency_key, fingerprint, order.order_id)
            return deepcopy(order), True

    def get_order(self, order_id: str) -> Optional[Order]:
        with self._lock:
            order = self._orders.get(order_id)
            if order is None:
                return None
            if (order.status == "pending_payment"
                    and datetime.now(timezone.utc) >= datetime.fromisoformat(
                        order.expires_at.replace("Z", "+00:00"))):
                order.status = "expired"
            return deepcopy(order)

    def _build_fingerprint(self, items) -> dict:
        return {
            "items": sorted(
                [{"product_id": pid, "quantity": qty} for pid, qty in items],
                key=lambda x: x["product_id"],
            )
        }

    def _build_order(self, items, idempotency_key: str) -> Order:
        order_items = []
        subtotal_total = 0
        total_units = 0

        for product_id, quantity in items:
            product = get_product(product_id)
            if product is None:
                # No debería ocurrir: validado antes.
                raise ValidationError(f"Producto inexistente: {product_id}.")

            unit_price = product["price_clp"]
            subtotal = compute_subtotal(unit_price, quantity)
            subtotal_total += subtotal
            total_units += quantity

            order_items.append(
                OrderItem(
                    product_id=product_id,
                    name=product["name"],
                    quantity=quantity,
                    unit_price_clp=unit_price,
                    subtotal_clp=subtotal,
                )
            )

        now = datetime.now(timezone.utc)
        expires = now + timedelta(minutes=QUOTE_TTL_MINUTES)

        return Order(
            order_id=str(uuid.uuid4()),
            idempotency_key=idempotency_key,
            status="pending_payment",
            items=order_items,
            total_units=total_units,
            subtotal_clp=subtotal_total,
            service_fee_clp=SERVICE_FEE_CLP,
            total_clp=subtotal_total + SERVICE_FEE_CLP,
            currency="CLP",
            created_at=now.isoformat(timespec="seconds").replace("+00:00", "Z"),
            expires_at=expires.isoformat(timespec="seconds").replace("+00:00", "Z"),
        )


order_service = OrderService(idempotency_store)

"""
Utilidades monetarias. Todos los importes son enteros en CLP.

Nunca se usa float para dinero.
"""

# Límites para evitar abusos: 50 unidades por producto y 200 por pedido.
MAX_QUANTITY_PER_PRODUCT = 50
MAX_UNITS_PER_ORDER = 200


def to_clp(amount: int) -> int:
    """Valida que un valor sea un entero CLP no negativo."""
    if not isinstance(amount, int) or isinstance(amount, bool):
        raise TypeError("El importe debe ser un entero.")
    if amount < 0:
        raise ValueError("El importe no puede ser negativo.")
    return amount


def compute_subtotal(unit_price_clp: int, quantity: int) -> int:
    return to_clp(unit_price_clp * quantity)

"""
Validación de solicitudes de pedidos.
"""

from typing import List, Tuple
from uuid import UUID

from data.product_catalog import get_product
from utils.money import MAX_QUANTITY_PER_PRODUCT, MAX_UNITS_PER_ORDER

ALLOWED_ITEM_FIELDS = {"product_id", "quantity"}
ALLOWED_PAYLOAD_FIELDS = {"items"}


class ValidationError(Exception):
    """Error de validación de la solicitud."""

    def __init__(self, message: str, field: str = None):
        super().__init__(message)
        self.message = message
        self.field = field

    def to_dict(self) -> dict:
        payload = {"error": "validation_error", "message": self.message}
        if self.field:
            payload["field"] = self.field
        return payload


def validate_idempotency_key(raw_key: str) -> str:
    if not raw_key or not isinstance(raw_key, str):
        raise ValidationError("El header Idempotency-Key es obligatorio.", "Idempotency-Key")
    stripped = raw_key.strip()
    try:
        parsed = UUID(stripped)
        if str(parsed) != stripped.lower():
            raise ValueError
    except ValueError:
        raise ValidationError("Idempotency-Key debe ser un UUID válido.", "Idempotency-Key")
    return str(parsed)


def validate_payload(payload) -> List[Tuple[str, int]]:
    """
    Valida el payload de POST /api/orders.

    Devuelve una lista de tuplas (product_id, quantity) validadas
    y normalizadas.
    """
    if not isinstance(payload, dict):
        raise ValidationError("El cuerpo debe ser un objeto JSON.")

    unexpected = set(payload.keys()) - ALLOWED_PAYLOAD_FIELDS
    if unexpected:
        raise ValidationError(
            f"Campos no permitidos: {sorted(unexpected)}", "body"
        )

    items = payload.get("items")
    if not isinstance(items, list) or len(items) == 0:
        raise ValidationError("Debes incluir al menos un producto.", "items")

    seen_ids = set()
    total_units = 0
    result: List[Tuple[str, int]] = []

    for index, item in enumerate(items):
        if not isinstance(item, dict):
            raise ValidationError(
                f"El item {index} debe ser un objeto.", f"items[{index}]"
            )
        unexpected = set(item.keys()) - ALLOWED_ITEM_FIELDS
        if unexpected:
            raise ValidationError(
                f"Campos no permitidos en el item {index}: {sorted(unexpected)}",
                f"items[{index}]",
            )

        product_id = item.get("product_id")
        quantity = item.get("quantity")

        if not isinstance(product_id, str) or not product_id.strip():
            raise ValidationError(
                f"product_id inválido en item {index}.", f"items[{index}].product_id"
            )

        if isinstance(quantity, bool) or not isinstance(quantity, int):
            raise ValidationError(
                f"quantity debe ser un entero en item {index}.",
                f"items[{index}].quantity",
            )

        if quantity <= 0:
            raise ValidationError(
                f"quantity debe ser mayor que cero en item {index}.",
                f"items[{index}].quantity",
            )

        if quantity > MAX_QUANTITY_PER_PRODUCT:
            raise ValidationError(
                f"quantity excede el máximo por producto ({MAX_QUANTITY_PER_PRODUCT}).",
                f"items[{index}].quantity",
            )

        if product_id in seen_ids:
            raise ValidationError(
                f"product_id duplicado: {product_id}.", f"items[{index}].product_id"
            )
        seen_ids.add(product_id)

        product = get_product(product_id)
        if product is None:
            raise ValidationError(
                f"Producto inexistente: {product_id}.",
                f"items[{index}].product_id",
            )

        if not product["available"]:
            raise ValidationError(
                f"Producto no disponible: {product_id}.",
                f"items[{index}].product_id",
            )

        total_units += quantity
        result.append((product_id, quantity))

    if total_units > MAX_UNITS_PER_ORDER:
        raise ValidationError(
            f"El pedido excede el máximo de unidades ({MAX_UNITS_PER_ORDER}).",
            "items",
        )

    return result

import { Injectable } from '@angular/core';
import { HttpClient, HttpHeaders } from '@angular/common/http';
import { Observable } from 'rxjs';
import { timeout } from 'rxjs/operators';
import { environment } from '../../../environments/environment';
import { CartSelectionService } from '../cart-selection.service';

export interface OrderRequestItem {
  product_id: string;
  quantity: number;
}

export interface OrderItem extends OrderRequestItem {
  name: string;
  unit_price_clp: number;
  subtotal_clp: number;
}

export interface Order {
  order_id: string;
  idempotency_key: string;
  status: 'pending_payment' | 'expired' | 'cancelled';
  items: OrderItem[];
  total_units: number;
  subtotal_clp: number;
  service_fee_clp: number;
  total_clp: number;
  currency: 'CLP';
  created_at: string;
  expires_at: string;
}

@Injectable({ providedIn: 'root' })
export class OrdersService {
  private readonly baseUrl = environment.apiUrl;
  private fingerprint = '';
  private idempotencyKey = '';

  constructor(private http: HttpClient, cart: CartSelectionService) {
    cart.items$.subscribe(items => this.syncCart(items.map(item => ({
      product_id: item.product.id,
      quantity: item.quantity
    }))));
  }

  // El servicio sobrevive a la navegación: los reintentos conservan la clave.
  syncCart(items: OrderRequestItem[]): void {
    const fingerprint = JSON.stringify([...items].sort((a, b) =>
      a.product_id.localeCompare(b.product_id)));
    if (fingerprint !== this.fingerprint) {
      this.fingerprint = fingerprint;
      this.idempotencyKey = '';
    }
  }

  prepareOrder(items: OrderRequestItem[]): Observable<Order> {
    this.syncCart(items);
    if (!this.idempotencyKey) {
      this.idempotencyKey = crypto.randomUUID();
    }
    return this.http.post<Order>(`${this.baseUrl}/orders`, { items }, {
      headers: new HttpHeaders({ 'Idempotency-Key': this.idempotencyKey })
    }).pipe(timeout(15000));
  }

  getOrder(orderId: string): Observable<Order> {
    return this.http.get<Order>(
      `${this.baseUrl}/orders/${encodeURIComponent(orderId)}`
    ).pipe(timeout(15000));
  }
}

import { Component, OnDestroy, OnInit } from '@angular/core';
import { HttpErrorResponse } from '@angular/common/http';
import { Router } from '@angular/router';
import { Observable, Subject, TimeoutError } from 'rxjs';
import { finalize, take, takeUntil } from 'rxjs/operators';
import { CartItem, CartSelectionService } from '../../services/cart-selection.service';
import { Order, OrderRequestItem, OrdersService } from '../../services/orders/orders.service';

@Component({
  selector: 'app-checkout',
  templateUrl: './checkout.component.html',
  styleUrls: ['./checkout.component.scss'],
  standalone: false
})
export class CheckoutComponent implements OnInit, OnDestroy {
  items$!: Observable<CartItem[]>;
  totalUnits$!: Observable<number>;
  subtotal$!: Observable<number>;
  readonly serviceFee = 0;
  loading = false;
  errorMessage = '';
  order: Order | null = null;
  priceDifference = 0;
  private currentItems: CartItem[] = [];
  private cartFingerprint = '';
  private cartVersion = 0;
  private readonly destroy$ = new Subject<void>();

  constructor(
    private cart: CartSelectionService,
    private router: Router,
    private orders: OrdersService
  ) {}

  ngOnInit(): void {
    this.cart.isEmpty$.pipe(take(1), takeUntil(this.destroy$)).subscribe(empty => {
      if (empty) {
        this.router.navigate(['/carrito']);
      }
    });
    this.items$ = this.cart.items$;
    this.totalUnits$ = this.cart.totalUnits$;
    this.subtotal$ = this.cart.totalPriceClp$;
    // Descartar resultados anteriores cuando cambia el contenido del carrito.
    this.items$.pipe(takeUntil(this.destroy$)).subscribe(items => {
      this.currentItems = items;
      const payload = this.requestItems();
      this.orders.syncCart(payload);
      const fingerprint = JSON.stringify(payload);
      if (fingerprint !== this.cartFingerprint) {
        this.cartFingerprint = fingerprint;
        this.cartVersion++;
        this.order = null;
        this.priceDifference = 0;
        this.errorMessage = '';
      }
    });
  }

  prepareOrder(): void {
    if (this.loading || this.currentItems.length === 0) {
      return;
    }
    this.loading = true;
    this.errorMessage = '';
    const version = this.cartVersion;
    const estimatedTotal = this.currentItems.reduce((sum, item) => sum + item.subtotalClp, 0)
      + this.serviceFee;
    this.orders.prepareOrder(this.requestItems()).pipe(
      takeUntil(this.destroy$),
      finalize(() => { this.loading = false; })
    ).subscribe({
      next: order => {
        if (version !== this.cartVersion) {
          return;
        }
        this.order = order;
        this.priceDifference = order.total_clp - estimatedTotal;
      },
      error: (error: unknown) => {
        if (version !== this.cartVersion) {
          return;
        }
        if (error instanceof TimeoutError ||
            (error instanceof HttpErrorResponse && error.status === 0)) {
          this.errorMessage = 'No se pudo conectar con el servidor. Reintenta: conservaremos la misma solicitud para evitar duplicados.';
        } else if (error instanceof HttpErrorResponse && error.status < 500) {
          this.errorMessage = error.error?.message || 'No se pudo validar el pedido. Revisa el carrito.';
        } else {
          this.errorMessage = 'El servidor no pudo preparar el pedido. Puedes reintentar.';
        }
      }
    });
  }

  ngOnDestroy(): void {
    this.destroy$.next();
    this.destroy$.complete();
  }

  formatClp(value: number): string {
    return this.cart.formatClp(value);
  }

  totalWithFee(subtotal: number): number {
    return subtotal + this.serviceFee;
  }

  goBack(): void {
    this.router.navigate(['/carrito']);
  }

  trackByProductId(_: number, item: CartItem): string {
    return item.product.id;
  }

  private requestItems(): OrderRequestItem[] {
    return this.currentItems.map(item => ({
      product_id: item.product.id,
      quantity: item.quantity
    })).sort((a, b) => a.product_id.localeCompare(b.product_id));
  }
}

ORDER_STATUSES = {"pending_payment", "expired", "cancelled"}


def utc_now_iso() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")


@dataclass
class OrderItem:
    product_id: str
    name: str
    quantity: int
    unit_price_clp: int
    subtotal_clp: int

    def to_dict(self) -> dict:
        return {
            "product_id": self.product_id,
            "name": self.name,
            "quantity": self.quantity,
            "unit_price_clp": self.unit_price_clp,
            "subtotal_clp": self.subtotal_clp,
        }


@dataclass
class Order:
    order_id: str
    idempotency_key: str
    status: str
    items: List[OrderItem]
    total_units: int
    subtotal_clp: int
    service_fee_clp: int
    total_clp: int
    currency: str
    created_at: str
    expires_at: str

    def to_dict(self) -> dict:
        return {
            "order_id": self.order_id,
            "idempotency_key": self.idempotency_key,
            "status": self.status,
            "items": [item.to_dict() for item in self.items],
            "total_units": self.total_units,
            "subtotal_clp": self.subtotal_clp,
            "service_fee_clp": self.service_fee_clp,
            "total_clp": self.total_clp,
            "currency": self.currency,
            "created_at": self.created_at,
            "expires_at": self.expires_at,
        }

    def to_fingerprint(self) -> Dict:
        """Huella del contenido relevante para idempotencia."""
        return {
            "items": [
                {"product_id": item.product_id, "quantity": item.quantity}
                for item in sorted(self.items, key=lambda i: i.product_id)
            ]
        }

"""
Almacén de idempotencia en memoria.

ADVERTENCIA — NO USAR EN PRODUCCIÓN:
- No resiste reinicios del proceso.
- No funciona con múltiples workers o réplicas.
- No tiene persistencia transaccional.
En producción debe sustituirse por una tabla con restricción UNIQUE
sobre idempotency_key y transacciones atómicas.
"""

import threading
from typing import Dict, Optional, Tuple


class IdempotencyStore:
    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._by_key: Dict[str, Tuple[dict, str]] = {}

    def get(self, key: str) -> Optional[Tuple[dict, str]]:
        """Devuelve (fingerprint, order_id) si existe la clave."""
        with self._lock:
            return self._by_key.get(key)

    def try_register(self, key: str, fingerprint: dict, order_id: str) -> bool:
        """
        Intenta registrar una clave con su fingerprint y order_id.
        Devuelve True si se registró; False si ya existía.
        """
        with self._lock:
            if key in self._by_key:
                return False
            self._by_key[key] = (fingerprint, order_id)
            return True

    def clear(self) -> None:
        """Solo para tests."""
        with self._lock:
            self._by_key.clear()


idempotency_store = IdempotencyStore()
